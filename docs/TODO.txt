Quest 3의 Inside-Out Body Tracking(IOBT) 기능을 사용해 단순 수학 공식이 아닌 실제 카메라 센서가 추적한 진짜 팔꿈치(Elbow) 데이터를 WebXR에서 가져오려면, Meta의 전용 WebXR 확장 API인 Meta Body Tracking Extension을 사용해야 합니다.현시점 WebXR 표준에는 Body Tracking이 완전히 통합되어 있지 않아서, Meta Quest 브라우저 전용 meta-body-tracking 기능(Feature)을 요청하는 방식으로 구현합니다.1. 구현 가이드라인 (JS 코드 수정)기존의 수학 계산 수식을 제거하고, Meta Body Tracking API로 바꾸는 전체 절차입니다.Step 1. WebXR 세션 생성 시 meta-body-tracking 옵션 요청navigator.xr.requestSession을 부를 때 optionalFeatures에 meta-body-tracking을 넣어줍니다.JavaScriptfunction onStart() {
  navigator.xr.requestSession("immersive-ar", {
    // meta-body-tracking 기능 요청 추가
    optionalFeatures: ["hand-tracking", "local-floor", "bounded-floor", "meta-body-tracking"]
  }).then(onSessionStart).catch((err) => {
    // 기능 미지원 시 기본 세션으로 폴백
    navigator.xr.requestSession("immersive-ar").then(onSessionStart);
  });
}
Step 2. Body Space 및 Joint Pose 가져오기세션이 열리면 XRBodySpaces 객체를 통해 관절(Joint) 데이터를 읽어올 준비를 합니다.JavaScriptlet bodySpaces = null;

function onSessionStart(session) {
  // ... 기존 코드 ...

  // Quest 브라우저에서 Body Tracking 지원 여부 확인 및 Body Spaces 가져오기
  if (session.enabledFeatures && session.enabledFeatures.includes("meta-body-tracking")) {
    // Meta 확장을 통해 관절 공간(Joint Spaces) 맵을 획득
    bodySpaces = session.requestBodySpaces ? session.requestBodySpaces() : null;
  }
  
  // ... 기존 requestAnimationFrame 등록 ...
}
Step 3. sendFrame 함수 내부에서 팔꿈치/어깨 관절 데이터 추출기존의 shoulder_X = headPos.x + ... 같은 가짜 수학 수식을 모두 삭제하고, 센서에서 받아온 실제 관절 좌표로 대체합니다.JavaScriptfunction sendFrame(session, space, time, frame) {
  // ... 기존 HMD 및 Controller 추출 코드 ...

  // === Inside-Out Body Tracking 센서 데이터 가져오기 ===
  if (bodySpaces && frame.getBodyPose) {
    const bodyPose = frame.getBodyPose(space);
    
    if (bodyPose) {
      for (const side of ["left", "right"]) {
        // Meta Body Tracking 관절 규격 이름
        const elbowJointName = `${side}-elbow`; // 'left-elbow', 'right-elbow'
        const shoulderJointName = `${side}-shoulder`; // 'left-shoulder', 'right-shoulder'

        // 1. 실제 팔꿈치(Elbow) 센서 포즈
        if (bodySpaces[elbowJointName]) {
          const elbowPose = frame.getJointPose(bodySpaces[elbowJointName], space);
          if (elbowPose) {
            response[`elbow_${side}`] = {
              x: elbowPose.transform.position.x,
              y: elbowPose.transform.position.y,
              z: elbowPose.transform.position.z,
              qx: elbowPose.transform.orientation.x,
              qy: elbowPose.transform.orientation.y,
              qz: elbowPose.transform.orientation.z,
              qw: elbowPose.transform.orientation.w,
            };
          }
        }

        // 2. 실제 어깨(Shoulder) 센서 포즈
        if (bodySpaces[shoulderJointName]) {
          const shoulderPose = frame.getJointPose(bodySpaces[shoulderJointName], space);
          if (shoulderPose) {
            response[`shoulder_${side}`] = {
              x: shoulderPose.transform.position.x,
              y: shoulderPose.transform.position.y,
              z: shoulderPose.transform.position.z,
              qx: shoulderPose.transform.orientation.x,
              qy: shoulderPose.transform.orientation.y,
              qz: shoulderPose.transform.orientation.z,
              qw: shoulderPose.transform.orientation.w,
            };
          }
        }
      }
    }
  }

  // 데이터 전송
  websocket.send(JSON.stringify(response));
}
